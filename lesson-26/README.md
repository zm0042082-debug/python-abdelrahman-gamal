## ماذا تعلمت؟

-  (#1|#2) & (#1.union(#2)) مع بعض بعمل (set) لو عايز ادمج جملتين 

EX: A={1,2,3}
    B={4,5,6}
    print(A.union(B))   &  print(A|B)
   ={1,2,3,4,5,6} 


-  #1.add(العنصر ال هضيفه)  بعمل  (set) لو عايز اضيف عنصر ع جملة  

EX: A={1,2,3}
    A.add(4)
    print(A)
   ={1,2,3,4} 


-   #1.add(العنصر التاني هضيفه)  #1.add(العنصر ال هضيفه)  بعمل   (set) لو عايز اضيف عنصرين ع جملة  

EX1: A={1,2,3}
    A.add(4)
    A.add(5)
    print(A)
   ={1,2,3,4,5}   (صح)   

EX2: A={1,2,3}
    A.add(4,5)
    A.add(6,7)
    print(A)
   = Type Error   (خطاء)

- #1.remove(العنصر ال هحذفه)  بعمل  (set) لو عايز احذف عنصر ف جملة  

EX1: A={1,2,3,"Zeyad"}
     A.remove("Zeyad")
     print(A)
    ={1,2,3} 
 
Type Error لكن لو حذفت عنصر وهو مش موجود ف الجمله هيديني 
 
EX2: A={1,2,3,"Zeyad"}
     A.remove(15)
     print(A)
    = Type Error 
 

- #1.discard(العنصر ال هحذفه)  بعمل  (set) لو عايز احذف عنصر ف جملة

EX1: A={1,2,3,"Zeyad"}
     A.discard("Zeyad")
     print(A)
    ={1,2,3} 
 
   لكن لو حذفت عنصر وهو مش موجود ف الجمله هيطبعها عادي
   
EX2: A={1,2,3,"Zeyad"}
     A.discard(15)
     print(A)
    ={1,2,3,"Zeyad"}

- #1.clear()  بعمل  (set) لو عايز احذف عنصر ف جملة

EX:  A={1,2,3,"Zeyad"}
     A.clear()
     print(A)
    = set()

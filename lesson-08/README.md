# اسم الدرس

concatenation ريط النصوص

## ماذا تعلمت؟

- :يمكن ربط النوص بهذه الطريقه
EX 1: name1= "Zeyad"
      name2= "Mahmoud"
      print (name1 + name2) 
      =ZeyadMahmoud

EX 2: name1= "Zeyad "
      name2= "Mahmoud"
      print (name1 + name2) 
      =Zeyad Mahmoud

EX 3: name1= "Zeyad"
      name2= "Mahmoud"
      print (name1+ " " +name2) 
      =Zeyad Mahmoud
      
EX 4: name1= "Zeyad"
      name2= "Mahmoud"
      names= name1 + name2
      print (names) 
      =ZeyadMahmoud

EX 5: name1= "Zeyad "
      name2= "Mahmoud"
      print ("Zeyad"+"Mahmoud")
      =ZeyadMahmoud
